1. What is an API?

API = Application Programming Interface. It’s a set of rules that lets two software programs talk to each other. Think of it like a waiter in a restaurant:

  • You (the client) look at the menu and tell the waiter what you want
  • The waiter (the API) takes your order to the kitchen
  • The kitchen (the server) prepares your food
  • The waiter brings back your food (the response)

You never go into the kitchen directly — the waiter is the interface.

3. Building a Simple REST API (PHP)

Goal: A complete CRUD API for managing contacts

This example uses plain PHP with no frameworks. It handles GET, POST, PUT, DELETE with JSON.

Step 1: Create the endpoint file

Save this as api.php and put it in your web root:

<?php
// api.php - Simple REST API for contacts
// No framework needed!

// Set JSON response header
header('Content-Type: application/json');

// --- DATABASE CONNECTION ---
function db() {
    static $pdo = null;
    if ($pdo === null) {
        $pdo = new PDO(
            'mysql:host=localhost;dbname=myapp;charset=utf8mb4',
            'root', '',
            [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]
        );
    }
    return $pdo;
}

// --- PARSE THE REQUEST ---
$method = $_SERVER['REQUEST_METHOD'];
$path   = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH);
$segments = array_filter(explode('/', $path));
$segments = array_values($segments);

// Extract ID if present: /api.php/contacts/5
$id = $segments[1] ?? null;

// Read JSON body if present
$input = json_decode(file_get_contents('php://input'), true);

// --- ROUTE THE REQUEST ---
try {
    if ($method === 'GET' && !$id) {
        // GET all contacts
        $rows = db()->query("SELECT * FROM contacts ORDER BY id")->fetchAll();
        echo json_encode(['success' => true, 'data' => $rows]);

    } elseif ($method === 'GET' && $id) {
        // GET one contact
        $stmt = db()->prepare("SELECT * FROM contacts WHERE id = ?");
        $stmt->execute([$id]);
        $row = $stmt->fetch();
        if ($row) {
            echo json_encode(['success' => true, 'data' => $row]);
        } else {
            http_response_code(404);
            echo json_encode(['success' => false, 'error' => 'Not found']);
        }

    } elseif ($method === 'POST') {
        // CREATE a new contact
        $stmt = db()->prepare(
            "INSERT INTO contacts (name, email, phone) VALUES (?, ?, ?)"
        );
        $stmt->execute([
            $input['name'],
            $input['email'],
            $input['phone'] ?? null
        ]);
        $newId = db()->lastInsertId();
        http_response_code(201);
        echo json_encode(['success' => true, 'id' => $newId]);

    } elseif ($method === 'PUT' && $id) {
        // UPDATE a contact
        $stmt = db()->prepare(
            "UPDATE contacts SET name=?, email=?, phone=? WHERE id=?"
        );
        $stmt->execute([
            $input['name'],
            $input['email'],
            $input['phone'] ?? null,
            $id
        ]);
        echo json_encode(['success' => true]);

    } elseif ($method === 'DELETE' && $id) {
        // DELETE a contact
        $stmt = db()->prepare("DELETE FROM contacts WHERE id = ?");
        $stmt->execute([$id]);
        echo json_encode(['success' => true]);

    } else {
        http_response_code(400);
        echo json_encode(['success' => false, 'error' => 'Invalid request']);
    }
} catch (Exception $e) {
    http_response_code(500);
    echo json_encode(['success' => false, 'error' => $e->getMessage()]);
}

Step 2: Create the database table

CREATE DATABASE myapp;
USE myapp;

CREATE TABLE contacts (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(200) NOT NULL,
    phone VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Step 3: Test it with curl

4. Building an Advanced API

Authentication: API Keys

The simplest auth method. Client sends a key in headers:

<?php
// Check API key at the top of your API file
$apiKey = $_SERVER['HTTP_X_API_KEY'] ?? '';

if ($apiKey !== 'your-secret-key-here') {
    http_response_code(401);
    echo json_encode(['error' => 'Invalid API key']);
    exit;
}

Client sends: curl -H "X-API-Key: your-secret-key-here" http://localhost/api.php

Authentication: JWT Tokens

JWT = JSON Web Token. A signed token that contains user info. Three parts separated by dots:

Figure 3: JWT token = Header + Payload + Signature

<?php
// --- JWT Helper Functions ---

function jwt_encode($payload, $secret, $expiry = 3600) {
    $header = json_encode(['alg' => 'HS256', 'typ' => 'JWT']);
    $payload['exp'] = time() + $expiry;
    $payload = json_encode($payload);

    $base64Header = rtrim(strtr(base64_encode($header), '+/', '-_'), '=');
    $base64Payload = rtrim(strtr(base64_encode($payload), '+/', '-_'), '=');
    $signature = hash_hmac('sha256', "$base64Header.$base64Payload", $secret, true);
    $base64Sig = rtrim(strtr(base64_encode($signature), '+/', '-_'), '=');

    return "$base64Header.$base64Payload.$base64Sig";
}

function jwt_decode($token, $secret) {
    [$header, $payload, $sig] = explode('.', $token);
    // Verify signature
    $expected = hash_hmac('sha256', "$header.$payload", $secret, true);
    $expected64 = rtrim(strtr(base64_encode($expected), '+/', '-_'), '=');
    if ($sig !== $expected64) return false;

    $data = json_decode(base64_decode($payload), true);
    if ($data['exp'] < time()) return false; // expired
    return $data;
}

// --- Usage ---
// Login endpoint: verify user, then issue token
$token = jwt_encode(['user_id' => 123, 'role' => 'admin'], 'your-secret');
// Client stores token, sends: Authorization: Bearer <token>

Rate Limiting

Prevent abuse by limiting requests per user/IP:

<?php
// Simple rate limiter using file-based storage
function rate_limit($key, $maxRequests = 60, $windowSeconds = 60) {
    $file = "/tmp/rate_" . md5($key);
    $now = time();

    if (file_exists($file)) {
        $data = json_decode(file_get_contents($file), true);
        // Remove old entries outside the window
        $data = array_filter($data, fn($t) => $t > $now - $windowSeconds);
    } else {
        $data = [];
    }

    if (count($data) >= $maxRequests) {
        http_response_code(429);
        header('Retry-After: ' . $windowSeconds);
        echo json_encode(['error' => 'Rate limit exceeded']);
        exit;
    }

    $data[] = $now;
    file_put_contents($file, json_encode($data));
}

Pagination

Don’t return all records at once. Use page and limit:

<?php
$page  = max(1, (int)($_GET['page'] ?? 1));
$limit = min(100, max(1, (int)($_GET['limit'] ?? 20)));
$offset = ($page - 1) * $limit;

// Get total count
$total = db()->query("SELECT COUNT(*) FROM contacts")->fetchColumn();

// Get page of data
$stmt = db()->prepare("SELECT * FROM contacts LIMIT ? OFFSET ?");
$stmt->execute([$limit, $offset]);

echo json_encode([
    'data'       => $stmt->fetchAll(),
    'page'       => $page,
    'limit'      => $limit,
    'total'      => $total,
    'total_pages' => ceil($total / $limit),
]);

Input Validation

<?php
function validate($input, $rules) {
    $errors = [];
    foreach ($rules as $field => $rule) {
        if ($rule['required'] && empty($input[$field])) {
            $errors[$field] = "$field is required";
            continue;
        }
        if (isset($input[$field])) {
            if (isset($rule['max']) && strlen($input[$field]) > $rule['max']) {
                $errors[$field] = "$field must be under {$rule['max']} chars";
            }
            if ($rule['type'] === 'email' && !filter_var($input[$field], FILTER_VALIDATE_EMAIL)) {
                $errors[$field] = "$field must be a valid email";
            }
        }
    }
    return empty($errors) ? true : $errors;
}

// Usage
$result = validate($input, [
    'name'  => ['required' => true, 'max' => 100],
    'email' => ['required' => true, 'type' => 'email'],
]);
if ($result !== true) {
    http_response_code(400);
    echo json_encode(['errors' => $result]);
    exit;
}

5. SNMP — Simple Network Management Protocol

SNMP is used to monitor and manage network devices like routers, switches, printers, and servers. It lets a central system (NMS) query devices for stats like CPU usage, bandwidth, uptime, and temperature.

Figure 4: SNMP manager queries agents; agents send traps for alerts

Practical Commands

# Install SNMP tools (Linux)
sudo apt install snmp snmp-mibs-downloader

# GET - Query a single value
snmpget -v2c -c public 192.168.1.1 1.3.6.1.2.1.1.1.0
# Returns: system description (e.g. "Linux server 5.15.0...")

# WALK - Get all values in a subtree
snmpwalk -v2c -c public 192.168.1.1 1.3.6.1.2.1.1
# Returns: sysDescr, sysObjectID, sysUpTime, sysContact, sysName...

# GETNEXT - Get next OID in the tree
snmpgetnext -v2c -c public 192.168.1.1 1.3.6.1.2.1.1.1

# SET - Write a value (requires write access)
snmpset -v2c -c private 192.168.1.1 1.3.6.1.2.1.1.5.0 s "newhostname"

# SNMP v3 (secure)
snmpget -v3 -l authPriv -u admin -A SHA -a authpass -x AES -X privpass 192.168.1.1 1.3.6.1.2.1.1.1.0

Send Email with PHP

<?php
// Simple PHP mail()
mail('recipient@example.com', 'Subject', 'Body text here');

// Better: PHPMailer library
use PHPMailer\PHPMailer\PHPMailer;

$mail = new PHPMailer(true);
$mail->isSMTP();
$mail->Host       = 'smtp.gmail.com';
$mail->SMTPAuth   = true;
$mail->Username   = 'you@gmail.com';
$mail->Password   = 'app-specific-password';
$mail->SMTPSecure = PHPMailer::ENCRYPTION_STARTTLS;
$mail->Port       = 587;

$mail->setFrom('you@gmail.com', 'Your Name');
$mail->addAddress('friend@example.com');
$mail->Subject = 'Test from PHPMailer';
$mail->Body    = '<h1>Hello!</h1><p>Sent via PHPMailer</p>';
$mail->isHTML(true);
$mail->send();

Practical Commands

# SFTP - interactive file transfer
sftp user@192.168.1.100
sftp> put localfile.txt          # upload
sftp> get remotefile.txt         # download
sftp> ls /var/www/               # list remote dir

# SCP - quick one-shot copy
scp file.txt user@server:/tmp/               # upload
scp user@server:/var/log/syslog ./           # download
scp -r ./mydir/ user@server:/backup/         # recursive copy

# rsync - intelligent sync (only copies changes)
rsync -avz ./site/ user@server:/var/www/site/        # sync dir
rsync -avz --delete ./site/ user@server:/var/www/    # mirror (deletes extra files)
rsync -avzP largefile.zip user@server:/backup/       # with progress + resume

# FTP (insecure, avoid if possible)
ftp 192.168.1.100
ftp> binary                    # set binary mode for non-text files
ftp> put file.zip
ftp> get file.zip

rsync is the best tool for backups

# rsync over SSH with bandwidth limit and dry-run
rsync -avz --dry-run --bwlimit=1000 -e ssh ./data/ user@server:/backup/

# Exclude patterns
rsync -avz --exclude='*.log' --exclude='.git' ./project/ user@server:/var/www/

# Schedule with cron (backup every night at 2am)
# Add to crontab: crontab -e
0 2 * * * rsync -avz /home/web/profolio/ /home/web/backup-$(date +\%Y\%m\%d)/

Firewall Rules (iptables / UFW)

# UFW (Uncomplicated Firewall) - Ubuntu/Debian
sudo ufw enable
sudo ufw default deny incoming
sudo ufw default allow outgoing
sudo ufw allow 22/tcp       # SSH
sudo ufw allow 80/tcp       # HTTP
sudo ufw allow 443/tcp      # HTTPS
sudo ufw status verbose

# iptables (more detailed control)
iptables -A INPUT -p tcp --dport 22 -j ACCEPT      # allow SSH
iptables -A INPUT -p tcp --dport 80 -j ACCEPT       # allow HTTP
iptables -A INPUT -p tcp --dport 443 -j ACCEPT      # allow HTTPS
iptables -A INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT
iptables -A INPUT -j DROP                            # drop everything else

SSH Security

# /etc/ssh/sshd_config hardening
PermitRootLogin no               # Don't allow root login
PasswordAuthentication no        # Keys only (no passwords)
Port 2222                        # Change default port
MaxAuthTries 3                   # Lock after 3 failed attempts
AllowUsers web admin             # Only specific users

# Generate SSH key pair
ssh-keygen -t ed25519 -C "your@email.com"

# Copy key to server
ssh-copy-id -p 2222 user@server

# SSH tunnel (forward local port through remote server)
ssh -L 3306:localhost:3306 user@remote-server
# Now localhost:3306 connects to remote MySQL

# SSH jump host (bastion)
ssh -J jumpuser@bastion-server user@internal-server
